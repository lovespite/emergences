add_library(emergences_slime_mold STATIC SlimeMoldScene.cpp SlimeMoldGpu.cpp)
add_library(emergences::slime_mold ALIAS emergences_slime_mold)
target_link_libraries(emergences_slime_mold PUBLIC emergences::sim)
emergences_target_defaults(emergences_slime_mold)

if(BUILD_TESTING)
    set(header_checks)
    foreach(header SlimeMoldScene SlimeMoldGpu SlimeMoldModel)
        set(check "${CMAKE_CURRENT_BINARY_DIR}/header_checks/${header}.cpp")
        file(GENERATE OUTPUT "${check}" CONTENT "#include \"scenarios/slime_mold/${header}.h\"\n")
        list(APPEND header_checks "${check}")
    endforeach()
    add_library(emergences_slime_mold_header_check OBJECT ${header_checks})
    target_link_libraries(emergences_slime_mold_header_check PRIVATE emergences::slime_mold)
    emergences_target_defaults(emergences_slime_mold_header_check)
    foreach(suite Scene Backend)
        string(TOLOWER "${suite}" name)
        add_executable(emergences_slime_mold_${name}_tests tests/SlimeMold${suite}Tests.cpp)
        target_link_libraries(emergences_slime_mold_${name}_tests PRIVATE emergences::slime_mold)
        emergences_target_defaults(emergences_slime_mold_${name}_tests)
        add_test(NAME slime_mold_${name} COMMAND emergences_slime_mold_${name}_tests)
        set_tests_properties(slime_mold_${name} PROPERTIES LABELS "cpu;sim;slime_mold" TIMEOUT 60)
    endforeach()
endif()
