if(BUILD_TESTING)
    set(header_checks)
    foreach(header ParticleLifeScene ParticleLifeGpu)
        set(check "${CMAKE_CURRENT_BINARY_DIR}/header_checks/${header}.cpp")
        file(GENERATE OUTPUT "${check}" CONTENT "#include \"scenarios/particle_life/${header}.h\"\n")
        list(APPEND header_checks "${check}")
    endforeach()
    add_library(emergences_particle_life_header_check OBJECT ${header_checks})
    target_link_libraries(emergences_particle_life_header_check PRIVATE emergences::particle_life)
    emergences_target_defaults(emergences_particle_life_header_check)

    add_executable(emergences_particle_life_scene_tests tests/ParticleLifeSceneTests.cpp)
    target_link_libraries(emergences_particle_life_scene_tests PRIVATE emergences::particle_life)
    emergences_target_defaults(emergences_particle_life_scene_tests)
    add_test(NAME particle_life_scene COMMAND emergences_particle_life_scene_tests)
    set_tests_properties(particle_life_scene PROPERTIES LABELS "cpu;sim;particle_life" TIMEOUT 60)

    add_executable(emergences_particle_life_backend_tests tests/ParticleLifeBackendTests.cpp)
    target_link_libraries(emergences_particle_life_backend_tests PRIVATE emergences::particle_life)
    emergences_target_defaults(emergences_particle_life_backend_tests)
    add_test(NAME particle_life_backend COMMAND emergences_particle_life_backend_tests)
    set_tests_properties(particle_life_backend PROPERTIES LABELS "cpu;sim;particle_life" TIMEOUT 30)
endif()
