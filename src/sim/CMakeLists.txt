add_library(emergences_sim STATIC World.cpp cpu/SpatialHash.cpp cpu/CpuSimulation.cpp)
add_library(emergences::sim ALIAS emergences_sim)
target_compile_features(emergences_sim PUBLIC cxx_std_20)
target_include_directories(emergences_sim PUBLIC "${PROJECT_SOURCE_DIR}/src")
target_link_libraries(emergences_sim PUBLIC glm::glm)
if(EMERGENCES_ENABLE_OPENMP)
    find_package(OpenMP QUIET COMPONENTS CXX)
endif()
if(EMERGENCES_ENABLE_OPENMP AND OpenMP_CXX_FOUND)
    target_link_libraries(emergences_sim PRIVATE OpenMP::OpenMP_CXX)
    target_compile_definitions(emergences_sim PUBLIC EMERGENCES_HAS_OPENMP=1)
else()
    target_compile_definitions(emergences_sim PUBLIC EMERGENCES_HAS_OPENMP=0)
    message(STATUS "CPU rule kernels will run serially (OpenMP disabled or unavailable)")
endif()
emergences_target_defaults(emergences_sim)

if(BUILD_TESTING)
    # Compile every public header independently. This also checks Agent's GPU ABI
    # and proves the simulation layer builds without Vulkan or GLFW installed.
    set(header_checks)
    foreach(header Agent Backend World Rule Intervention Simulation cpu/SpatialHash cpu/RuleKernel cpu/CpuSimulation)
        set(check "${CMAKE_CURRENT_BINARY_DIR}/header_checks/${header}.cpp")
        file(GENERATE OUTPUT "${check}" CONTENT "#include \"sim/${header}.h\"\n")
        list(APPEND header_checks "${check}")
    endforeach()
    add_library(emergences_sim_header_check OBJECT ${header_checks})
    target_link_libraries(emergences_sim_header_check PRIVATE emergences::sim)
    emergences_target_defaults(emergences_sim_header_check)

    add_executable(emergences_world_grid_tests tests/WorldGridTests.cpp)
    target_link_libraries(emergences_world_grid_tests PRIVATE emergences::sim)
    emergences_target_defaults(emergences_world_grid_tests)
    add_test(NAME sim_world_grid COMMAND emergences_world_grid_tests)

    add_executable(emergences_cpu_tests tests/CpuSimulationTests.cpp)
    target_link_libraries(emergences_cpu_tests PRIVATE emergences::sim)
    emergences_target_defaults(emergences_cpu_tests)
    add_test(NAME sim_cpu COMMAND emergences_cpu_tests)
    add_executable(emergences_cpu_parallel_tests tests/CpuParallelTests.cpp)
    target_link_libraries(emergences_cpu_parallel_tests PRIVATE emergences::sim)
    if(EMERGENCES_ENABLE_OPENMP AND OpenMP_CXX_FOUND)
        target_link_libraries(emergences_cpu_parallel_tests PRIVATE OpenMP::OpenMP_CXX)
    endif()
    emergences_target_defaults(emergences_cpu_parallel_tests)
    add_test(NAME sim_cpu_parallel COMMAND emergences_cpu_parallel_tests)
    set_tests_properties(sim_world_grid sim_cpu sim_cpu_parallel PROPERTIES LABELS "cpu;sim" TIMEOUT 30)
endif()
